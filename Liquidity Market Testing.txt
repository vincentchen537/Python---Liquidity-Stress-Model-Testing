import pandas as pd
import numpy as np
import matplotlib.pyplot as mpl
import random

pd.set_option('display.max_columns', None)

portfolio = pd.DataFrame({
      "asset": ["Equity A", "Equity B", "Corporate Bond A", "Treasury A", "EUR/USD", "Rates Swap A"],
      "asset class": ["Equity", "Equity", "Fixed Income", "Rates", "FX", "Rates"],
      "position value": [50_000_000, 30_000_000, 40_000_000, 100_000_000, 60_000_000, 80_000_000],
      "daily volume": [20_000_000, 10_000_000, 5_000_000, 200_000_000, 100_000_000, 50_000_000],
      "bid/ask spread bps": [10, 20, 40, 1, 2, 5]
})

#Portfolio: collection of financial positions

#Position value: how much money the position is worth

#Daily volume: how much of that asset normally trades in the market each day

#Market liquidity: The higher liquidity, the more it can be bought or sold without moving price too much

#Bid/ask spread: Bid - What someone is willing to pay; Ask - What someone wants you to pay to buy it; Spread - difference between bid/ask

#Basis point: 1/100 of a percentage point

#Liquidation: Selling/unwinding positions to turn them into cash under stressed market conditions

#Liquidation days: anticipated number of days to sell the position

#Participation rate: Measure of the size of our liquidation relative to daily volume

#Market impact: price movement caused by own trading activity

#Market impact percentage: use square root for a non-linear relationship

#Market impact parameter: How strong does participation translate to price impact as model parameter

#Liquidation cost: Combination of the costs of crossing bid/ask spread percentage in basis points and market impact associated with the participation rate

#Capital: money a bank has available to absorb losses to stay financially stable

#RWA: assets are assigned risk percentages to determine how risky they are and how much capital is needed to cover the risk

def calculate_liquidation_cost(portfolio, liquidation_days, y = 0.001):

    x = portfolio.copy()

    v_stress = ("stress volume"
                if "stress volume" in x.columns
                else "daily volume")

    s_stress = ("stress spread"
                if "stress spread" in x.columns
                else "bid/ask spread bps")

    x["daily liquidation"] = x["position value"] / liquidation_days

    x["participation rate"] = x["daily liquidation"] / x[v_stress]

    x["spread %"] = x[s_stress] / 10_000

    x["spread cost"] = x["position value"] * x["spread %"]

    x["market impact %"] = y * np.sqrt(x["participation rate"])

    x["market impact cost"] = x["position value"] * x["market impact %"]

    x["total liquidation cost"] = x["spread cost"] + x["market impact cost"]

    return x

def stress_scenarios(portfolio, volume_scale, spread_scale):

    f = portfolio.copy()

    f["stress volume"] = f["daily volume"] * volume_scale

    f["stress spread"] = f["bid/ask spread bps"] * spread_scale

    return f

liquidation_days = [1, 2, 5, 10, 20]

volume_scales = [1.2, 1, 0.8, 0.6, 0.4]

spread_scales = [0.75, 1, 1.25, 1.5, 2]

def liquidation_cost(liquidation_days, volume_scales, spread_scales):

    combined_results = []

    for a in liquidation_days:

        for b in volume_scales:

            for c in spread_scales:

                total_stress = stress_scenarios(portfolio, volume_scale = b, spread_scale = c)

                result = calculate_liquidation_cost(total_stress, liquidation_days = a)

                total_liquidation_cost = result["total liquidation cost"].sum()

                combined_results.append({"Liquidation Days": a, "Volume Scale": b, "Spread Scale": c, "Liquidation Cost": round(total_liquidation_cost, 2)})

    combined_results = pd.DataFrame(combined_results)

    return combined_results

results = liquidation_cost(liquidation_days, volume_scales, spread_scales)

print(results.to_string(index=False))

random_day = int(input("Enter days to liquidate assets over (1, 2, 5, 10, or 20 day period): "))

while random_day not in liquidation_days:

    random_day = int(input("Needs to be over a 1, 2, 5, 10, or 20 day period: "))

random_volume_scale = random.choice(volume_scales)

random_spread_scale = random.choice(spread_scales)

random_liquidation_cost = results[(results["Liquidation Days"] == random_day) & (results["Volume Scale"] == random_volume_scale) & (results["Spread Scale"] == random_spread_scale)]

chosen_liquidation_cost = float(random_liquidation_cost["Liquidation Cost"].iloc[0])

print(f"Liquidation cost for {100 * (random_volume_scale - 1)}% volume change and {100 * (random_spread_scale - 1)}% spread change over a {random_day} day period is {chosen_liquidation_cost}")

starting_cash = float(input("Enter starting cash: "))

misc_outflow = float(input("Enter other daily cash outflow: "))

total_position = portfolio["position value"].sum()

daily_assets = total_position / random_day

daily_liquidation = chosen_liquidation_cost / random_day

liquidity_results = []

print(f"Daily asset value: ${daily_assets: ,.2f}")

print(f"Daily liquidation cost: ${daily_liquidation: ,.2f}")

r = float(input("Enter collateral rate (as decimal): "))

collateral = total_position * r

print(f"Set collateral = ${collateral: ,.2f}")

liquidity = starting_cash

max_shortfall = 0

shortfall_day = None

for day in range(1, random_day + 1):

    beginning_cash = liquidity

    assets = daily_assets

    liquidation_cost = daily_liquidation

    other_outflow = misc_outflow

    collateral_outflow = collateral if day == 1 else 0

    liquidity = beginning_cash + assets - liquidation_cost - collateral_outflow - other_outflow

    liquidity_results.append({"Day": day, "Ending Cash": f"${liquidity: ,.2f}"})
        
    if liquidity < 0:

        if shortfall_day is None:
            
            shortfall_day = day

        max_shortfall = max(max_shortfall, abs(liquidity))

if shortfall_day is None:

    print(f"Having starting cash of ${starting_cash: ,.2f} a collateral of {100 * r}% is always safe")

ending_liquidity = liquidity

print(f"First shortfall day: {shortfall_day}")

print(f"Largest shortfall: ${max_shortfall:,.2f}")

print(f"Ending liquidity: ${ending_liquidity:,.2f}")

liquidity_results = pd.DataFrame(liquidity_results)

print(liquidity_results.to_string(index=False))

portfolio["risk weights"] = [1, 1, 1, 0.2, 0.5, 0.5]

portfolio["RWA"] = portfolio["position value"] * portfolio["risk weights"]

total_rwa = portfolio["RWA"].sum()

print(f"Total RWA: ${total_rwa:,.2f}")

capital_ratios = [0.08, 0.10, 0.12, 0.15]

risk_weight_scales = [0.75, 1, 1.25, 1.50]

capital_results = []

for ratio in capital_ratios:

    for rwa in risk_weight_scales:

        stressed_rwa = (portfolio["position value"] * portfolio["risk weights"] * rwa).sum()

        required_capital = stressed_rwa * ratio

        capital_results.append({"RWA Scale": f"{100 * rwa}%", "Capital Ratio": f"{100 * ratio}%", "RWA": f"${stressed_rwa: ,.2f}", "Required Capital": f"${required_capital: ,.2f}"})

capital_results = pd.DataFrame(capital_results)

print(capital_results.to_string(index=False))
